cmake_minimum_required(VERSION 3.22)
project(BlazerStudio VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Qt6 6.8 REQUIRED COMPONENTS Quick QuickControls2 Svg Test)
qt_standard_project_setup(REQUIRES 6.8)
qt_add_executable(blazer-studio src/main.cpp src/document.cpp src/document.h src/workspace.cpp src/workspace.h src/simulator.cpp src/simulator.h)
qt_add_qml_module(blazer-studio URI BlazerStudio VERSION 1.0 QML_FILES qml/Main.qml qml/ActionButton.qml qml/SectionLabel.qml qml/NumberField.qml qml/ManufacturerDialog.qml)
qt_add_resources(blazer-studio profiles PREFIX / FILES profiles/blaze-x-m3.json profiles/manufacturer-materials.json)
target_link_libraries(blazer-studio PRIVATE Qt6::Quick Qt6::QuickControls2 Qt6::Svg)
target_compile_options(blazer-studio PRIVATE -Wall -Wextra -Wpedantic)
include(CTest)
if(BUILD_TESTING)
 qt_add_executable(blazer-tests tests/core_test.cpp src/document.cpp src/document.h src/simulator.cpp src/simulator.h)
 target_include_directories(blazer-tests PRIVATE src)
 target_link_libraries(blazer-tests PRIVATE Qt6::Quick Qt6::Svg Qt6::Test)
 add_test(NAME core COMMAND blazer-tests -platform offscreen)
 qt_add_executable(blazer-ui-tests tests/ui_test.cpp src/document.cpp src/document.h src/workspace.cpp src/workspace.h src/simulator.cpp src/simulator.h)
 qt_add_resources(blazer-ui-tests ui_test_qml PREFIX /qt/qml/BlazerStudio FILES qml/Main.qml qml/ActionButton.qml qml/NumberField.qml qml/SectionLabel.qml qml/ManufacturerDialog.qml)
 target_include_directories(blazer-ui-tests PRIVATE src)
 target_compile_definitions(blazer-ui-tests PRIVATE BLAZER_SOURCE_DIR="${CMAKE_CURRENT_SOURCE_DIR}")
 target_link_libraries(blazer-ui-tests PRIVATE Qt6::Quick Qt6::QuickControls2 Qt6::Svg Qt6::Test)
 add_test(NAME ui COMMAND blazer-ui-tests -platform offscreen)
 set_tests_properties(core ui PROPERTIES ENVIRONMENT "QT_QUICK_BACKEND=software")
endif()
install(TARGETS blazer-studio RUNTIME DESTINATION bin)
